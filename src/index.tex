\begin{exer*}{\Large{\textbf{\hspace{3.86cm}Zigzag Subsequences}}}

\begin{multicols}{2}
A sequence $y_1, \dots, y_k$ is \textit{zigzag} if $k=1$ or if consecutive differences $y_{i+1} - y_i$ are nonzero and alternate in sign. Let $X_1, \dots, X_n$ be chosen i.i.d. from $[0, 1]$. Let $a(X_1, \dots, X_n)$ be the length of the longest zigzag subsequence, and $f(n)$ be its expected value.
\columnbreak
\begin{center}
\begin{tikzpicture}[scale=1, every node/.style={scale=0.8}]
    % Draw Axes
    \draw[->] (0,0) -- (5.5,0) node[right] {Index ($i$)};
    \draw[->] (0,0) -- (0,3.5) node[above] {Value ($X_i$)};
    \draw[dashed, gray!50] (0,3) -- (5,3) node[right, black] {1.0};
    \draw[dashed, gray!50] (0,0) node[left, black] {0.0};

    % Define Points (Randomized zigzag-like sequence)
    \coordinate (P1) at (1, 2.5);
    \coordinate (P2) at (2, 0.5);
    \coordinate (P3) at (3, 2.8);
    \coordinate (P4) at (4, 1.2);
    \coordinate (P5) at (5, 2.2);

    % Draw the zigzag path
    \draw[thick, blue] (P1) -- (P2) -- (P3) -- (P4) -- (P5);

    % Draw the points
    \foreach \i in {1,...,5} {
        \filldraw[black] (P\i) circle (2pt);
        % \node[below=5pt] at (\i, 0) {$i=\i$};
    }

    % Labels for points
    \node[above right] at (P1) {$X_1$};
    \node[below left]  at (P2) {$X_2$};
    \node[above]       at (P3) {$X_3$};
    \node[below]       at (P4) {$X_4$};
    \node[above]       at (P5) {$X_5$};
    % \node[below right] at (P6) {$X_6$};

    % Annotate signs of differences
    \node[blue, font=\small] at (1.3, 1.3) {$-$};
    \node[blue, font=\small] at (2.3, 1.8) {$+$};
    \node[blue, font=\small] at (3.4, 1.8) {$-$};
    \node[blue, font=\small] at (4.4, 1.9) {$+$};
    % \node[blue, font=\small] at (5.5, 1.3) {$-$};

    % Legend or caption-like note
    \node[draw, fill=white, font=\footnotesize, align=left] at (3, -1) {
        Consecutive differences: $(X_2-X_1 < 0), (X_3-X_2 > 0), (X_4-X_3 < 0) \dots$ \\
        Signs: $(-, +, -, +) \implies$ \textbf{Zigzag Sequence}
    };
\end{tikzpicture}
\end{center}
\end{multicols}
\paragraph{Part A} 
\begin{enumerate}[nosep, leftmargin=*]
    \item Show any two distinct numbers form a zigzag, hence $f(2) = 2$.
    \item Verify $f(3) = 8/3$ by averaging $a$ over all $3! = 6$ equally likely orderings of three numbers.
\end{enumerate}

\paragraph{Part B} For $n \ge 3$, let $\Delta_n = a(X_1, \dots, X_n) - a(X_2, \dots, X_n)$ be the drop in length when $X_1$ is removed.
\begin{enumerate}[label=(\roman*), nosep, leftmargin=*]
    \item Show that $\Delta_n$ is always $0$ or $1$.
    \item Show $\Delta_n = 1$ if and only if the triple $(X_1, X_2, X_3)$ is non-monotone.
\end{enumerate}
Conclude from Part A that $\Delta_n = 1$ occurs on average $2/3$ of the time for any $n \ge 3$.

\paragraph{Part C} Derive a formula for $f(n)$ using $f(2)=2$ as a base case.
\begin{enumerate}[label=(\roman*), nosep, leftmargin=*]
    \item Explain why $f(n) - f(n-1) = E[\Delta_n]$.
    \item Determine the final expression for $f(n)$.
\end{enumerate}


\end{exer*}
